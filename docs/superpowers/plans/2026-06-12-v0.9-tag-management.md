- [ ] **Step 8: Implement transactional update and merge**

Add `update_tag`. The important order is: validate, collect affected IDs, mutate relations/name, refresh each affected asset, load result, commit.

```rust
pub async fn update_tag(
    db: &Db,
    tag_id: i64,
    name: &str,
    color: &str,
) -> anyhow::Result<crate::models::Tag> {
    let name = normalize_tag_name(name);
    anyhow::ensure!(!name.is_empty(), "tag name cannot be empty");
    let color = crate::tags::normalize_tag_color(color)?;
    let mut tx = db.begin().await?;
    let source = load_tag(&mut *tx, tag_id).await?;
    let affected_ids: Vec<i64> = sqlx::query_scalar(
        "SELECT asset_id FROM asset_tags WHERE tag_id = ?1 ORDER BY asset_id"
    )
    .bind(tag_id)
    .fetch_all(&mut *tx)
    .await?;

    let target_id: Option<i64> =
        sqlx::query_scalar("SELECT id FROM tags WHERE name = ?1 AND id != ?2")
            .bind(&name)
            .bind(tag_id)
            .fetch_optional(&mut *tx)
            .await?;

    let result_id = if let Some(target_id) = target_id {
        sqlx::query(
            "INSERT OR IGNORE INTO asset_tags (asset_id, tag_id)
             SELECT asset_id, ?1 FROM asset_tags WHERE tag_id = ?2"
        )
        .bind(target_id)
        .bind(tag_id)
        .execute(&mut *tx)
        .await?;
        sqlx::query("DELETE FROM tags WHERE id = ?1")
            .bind(tag_id)
            .execute(&mut *tx)
            .await?;
        target_id
    } else {
        sqlx::query(
            "UPDATE tags
             SET name = ?1, color = ?2, last_used_at = ?3
             WHERE id = ?4"
        )
        .bind(&name)
        .bind(&color)
        .bind(Utc::now().to_rfc3339())
        .bind(tag_id)
        .execute(&mut *tx)
        .await?;
        tag_id
    };

    for asset_id in affected_ids {
        refresh_asset_search_document(&mut *tx, asset_id).await?;
    }
    let result = load_tag(&mut *tx, result_id).await?;
    tx.commit().await?;
    let _ = source;
    Ok(result)
}
```

- [ ] **Step 9: Implement batch unlink and deletion**

Use `QueryBuilder<Sqlite>` rather than interpolating IDs:

```rust
pub async fn remove_tag_from_assets(
    db: &Db,
    tag_id: i64,
    asset_ids: &[i64],
) -> anyhow::Result<u64> {
    anyhow::ensure!(!asset_ids.is_empty(), "asset ids cannot be empty");
    let mut tx = db.begin().await?;
    load_tag(&mut *tx, tag_id).await?;

    let mut builder = sqlx::QueryBuilder::<sqlx::Sqlite>::new(
        "DELETE FROM asset_tags WHERE tag_id = "
    );
    builder.push_bind(tag_id);
    builder.push(" AND asset_id IN (");
    let mut separated = builder.separated(", ");
    for asset_id in asset_ids {
        separated.push_bind(asset_id);
    }
    separated.push_unseparated(")");
    let removed = builder.build().execute(&mut *tx).await?.rows_affected();

    for asset_id in asset_ids {
        refresh_asset_search_document(&mut *tx, *asset_id).await?;
    }
    tx.commit().await?;
    Ok(removed)
}

pub async fn delete_tag(db: &Db, tag_id: i64) -> anyhow::Result<bool> {
    let mut tx = db.begin().await?;
    let affected_ids: Vec<i64> = sqlx::query_scalar(
        "SELECT asset_id FROM asset_tags WHERE tag_id = ?1 ORDER BY asset_id"
    )
    .bind(tag_id)
    .fetch_all(&mut *tx)
    .await?;
    let deleted = sqlx::query("DELETE FROM tags WHERE id = ?1")
        .bind(tag_id)
        .execute(&mut *tx)
        .await?
        .rows_affected() == 1;
    if !deleted {
        tx.rollback().await?;
        return Ok(false);
    }
    for asset_id in affected_ids {
        refresh_asset_search_document(&mut *tx, asset_id).await?;
    }
    tx.commit().await?;
    Ok(true)
}
```

## Acceptance Criteria

- 标签列表显示准确的资源数量。

- 单选资源可以解除已有标签。
- 多选资源可以解除共同标签。
- 解除标签只删除关系，不删除标签。
- 删除标签需要明确二次确认，只删除应用内标签及关系。
- 删除标签不删除资产索引，不触碰原始素材文件。
- 标签操作后标签管理器、详情面板、最近标签、搜索结果和数量保持一致。
- `npm test`、`npm run build`、`cargo test`、`cargo check` 全部通过。
- 所有版本元数据一致为 `0.9.0`。

## Self-Review

- **Spec coverage:** 覆盖已确认规格中的名称、颜色、计数、合并、安全删除、单选解除、多选解除、FTS 和状态刷新。
- **Scope check:** 单一标签子系统，不引入标签层级、AI、导入导出、云同步或源文件编辑。
- **Placeholder scan:** 不含 `TBD`、`TODO`、模糊错误处理或未定义接口。

- **Safety check:** 唯一删除对象为 SQLite `tags` 和 `asset_tags`；不执行文件删除命令，不修改原始素材文件。
